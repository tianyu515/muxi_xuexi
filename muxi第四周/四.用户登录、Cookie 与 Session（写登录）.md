目的：定义用户登录函数，与可以拿到用户信息的函数
```go
type Rep struct{
    Username string `json:"username"`
    Passport string `json:"passport"`
}
var sessions = make(map[string]string)
var sessionsMu sync.Mutex
func loginRequest(w http.ResponseWriter,r *http.Request) {
    w.Header().Set("Content-Type","application/json,charset=utf-8")
    if r.Method != http.MethodPost {
        w.Header().Set("Content-Type","application/json,charset=utf-8")
        w.WriteHeader(http.StatusMethodNotAllowed)
        json.NewEncoder(w).Encode(map[string]string{
            "message" : "请使用POST",
        })
         return
    }
    var rep Rep
    err := json.NewDecoder(r.Body).Decode(&rep)
    if err != nil {
        w.WriteHeader(http.StatusBadRequest)
        json.NewEncoder(w).Encode(map[string]string{
            "message" : "JSON解析失败",
        })
        return
    }
    usersMu.Lock()
    user,exists := users[rep.Username]
    usersMu.Unlock()
    //验证用户名和密码
    if !exists || rep.Passport != user.Passport   {
        w.WriteHeader(http.StatusUnauthorized)
        json.NewEncoder(w).Encode(map[string]string{
            "message" : "用户名或密码错误",
        })
        return
    }
    //设置session
    sessionId := rep.Username + "_sessiom"
    //保存session
    sessionsMu.Lock()
    sessions[sessionId] = user.Username
    sessionsMu.Unlock()
    //设置cookie
    http.SetCookie(w,&http.Cookie{
        Name : "session_id",
        Path : "/",
        Value : sessionId,
    })
    w.WriteHeader(http.StatusOK)
    json.NewEncoder(w).Encode(map[string]string{
        "message": "登录成功",
        "username": user.Username,
    })
}
//查询用户信息
func me(w http.ResponseWriter,r *http.Request) {
    w.Header().Set("Content-Type","application/json")
    //从请求中查找Cookie
    cookie,err := r.Cookie("session_id")
    if err != nil {
        w.WriteHeader(http.StatusUnauthorized)
        json.NewEncoder(w).Encode(map[string]string{
            "message" : "请先登录",
        })
        return
    }
    //查找session
    sessionsMu.Lock()
    username,exists := sessions[cookie.Value]
    if !exists {
        w.WriteHeader(http.StatusUnauthorized)
        json.NewEncoder(w).Encode(map[string]string{
            "message" : "登录状态失效",
        })
        return
    }
    //查找用户
    usersMu.Lock()
    user,exists := users[username]
    if !exists {
        w.WriteHeader(http.StatusUnauthorized)
        json.NewEncoder(w).Encode(map[string]string{
            "message" : "用户不存在",
        })
        return
    }
    //发送用户信息
    w.WriteHeader(http.StatusOK)
    json.NewEncoder(w).Encode(map[string]string{
        "Nickname": user.Nickname,
        "username": user.Username,
    })
}
```
- 可以正常登录，也可以正常返回cookie,为什么返回用户信息和改密码的时候却返回登录状态失效（sessions[cookie.Value],exists为false）,因为我的sessionId设的是：user.Username+"session_id",在sessionsMap中，key可以是中文。Json也可以传递中文，但是Cookie的Value有字符的限制，中文字节不属于Cookie字符，所以net/http发送Cookie的时候，会把中文字符给滤掉，导致在返回用户信息的时候，value改变了，找不到对应的登录状态
