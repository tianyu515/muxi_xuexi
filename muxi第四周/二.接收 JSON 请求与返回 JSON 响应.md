```go
package main
import(
    "fmt"
    "encoding/json"
    "net/http"
)
type User struct{
    Username string `json:"username"`
    Nickname string `json:"nickname"`
    Passport string `json:"passport"`
}
func register(w http.ResponseWriter,r *http.Request) {
    //检查请求方法
    if r.Method != http.MethodPost {
        w.Header().Set("Content-Type","application/json,charset=utf-8")
        w.WriteHeader(http.StatusMethodNotAllowed)
        json.NewEncoder(w).Encode(map[string]string{
            "message" : "请使用POST",
        })
         return
    }
    //检查json是否合规
    var user User
    err := json.NewDecoder(r.Body).Decode(&user)
    if err != nil {
        w.Header().Set("Content-Type","application/json,charset=utf-8")
        w.WriteHeader(http.StatusBadRequest)
        json.NewEncoder(w).Encode(map[string]string{
            "message" : "JSON解析错误",
        })
        return
    }
    //检查用户名字和密码是否为空
    if user.Username == "" || user.Passport == "" {
        w.Header().Set("Content-Type","application/json,charset=utf-8")
        w.WriteHeader(http.StatusBadRequest)
        json.NewEncoder(w).Encode(map[string]string{
            "message" : "用户名和密码不能为空",
        })
        return
    }
    //返回JSON
    w.Header().Set("Content-Type","application/json,charset=utf-8")
        w.WriteHeader(http.StatusCreated)
        json.NewEncoder(w).Encode(map[string]string{
            "message" : "注册成功",
            "username" : user.Username,
            "nickname" : user.Nickname,
        })
}
func main() {
    http.HandleFunc("/register",register)
    err := http.ListenAndServe(":8081",nil)
    if err != nil {
        fmt.Println("创建失败")
    }
}
```
- 在写Web服务中，自己定义的这个函数就显得很重要，用户访问路径的时候，全权交由该函数
- json.NewDecoder(r.Body).Decode(&user):NewDecoder可以直接读取数据流中的json信息，Decode将json格式解码后写在user变量里。要传一个指针变量进去，方便修改
  json.NewEncoder(w).Encode(map[string]string{})：将传进去的信息编码成json的形式，通过(w)写入响应体。**Encode接受的类型很多，但不能是函数**。像只是去传入一句话，map就很合适，不用再单独创建一个结构体
- 响应头的建立：w.Header().Set()
  响应情况的输入：w.WriteHeader()
- 服务端发信息不用接收错误，把客户端的信息转成json形式时有错误要接收，以免传入不符合规范的json形式
- 在register函数有三次检查和一次发送信息，每一次操作写返回状态码(**响应头的设置可以只用一次**)
